async function mineNewBlock() { const btn = document.getElementById('mineBtn'); btn.disabled = true; // Disable so you can't double-mine const index = chain.length; const prevHash = chain[chain.length - 1].hash; let nonce = 0; let hash = ""; // Create the visual block immediately const display = document.createElement('div'); display.className = "block mining"; document.getElementById('chain').prepend(display); // --- THE MINING LOOP --- while (true) { nonce++; hash = await sha256(index + prevHash + nonce); // Update UI every 100 tries so it doesn't lag if (nonce % 100 === 0) { display.innerHTML = `

⛏️ Mining Block #${index}...

Nonce: ${nonce}

${hash}
`; // This is the "Magic" line that prevents the browser from freezing await new Promise(resolve => setTimeout(resolve, 0)); } // Check if we found a valid hash (starts with 0000) if (hash.startsWith("0000")) { break; } } // Success! Update block style display.className = "block"; display.style.borderColor = "#00ff00"; display.innerHTML = `

✅ Block #${index} Mined!

Final Nonce: ${nonce}

Hash:

${hash}
`; chain.push({ index, hash, nonce }); btn.disabled = false; }
